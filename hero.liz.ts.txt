// 販売サイトの最初の面の背景（site/hero.mp4）。Liz CAP でこのまま書き出したもの。
// 単位円の上を回る点の高さが、右の sin のグラフになる。1 周 9 秒で、終わりが始まりにつながる。

const cx = 1.4, r = 1.3;          // 円の中心と半径
const x0 = 3.3, L = 3.5;          // sin のグラフの左端と 1 周期の長さ
const LINE = "#62666d", INK = "#d0d6e0", ACCENT = "#e4f222";

const ax = axes({ xRange: [0, TAU, PI / 2], yRange: [-1, 1, 1], width: L, height: 2 * r, x: x0 + L / 2, y: 0, color: LINE });
line({ from: [cx - r - 0.4, 0], to: [cx + r + 0.4, 0], color: LINE, width: 0.02 });
line({ from: [cx, -r - 0.4], to: [cx, r + 0.4], color: LINE, width: 0.02 });
circle({ x: cx, r, color: INK, strokeWidth: 0.03, opacity: 0.8 });
ax.plot((t) => Math.sin(t), { color: INK, width: 0.03 });
tex("y = \\sin\\theta", { x: x0 + L - 0.9, y: r + 0.6, size: 0.45, color: "#8a8f98" });

const p = dot({ x: cx + r, y: 0, r: 0.07, color: ACCENT });
const q = dot({ x: x0, y: 0, r: 0.07, color: ACCENT });
line({ from: [cx, 0], to: p, color: ACCENT, width: 0.025 });
line({ from: p, to: q, color: ACCENT, width: 0.015, opacity: 0.5 });

const N = 90;                      // 1 周を 90 歩に分ける（0.1 秒ずつ）
for (let k = 1; k <= N; k++) {
  const th = (k / N) * TAU;
  play(p.rotate(TAU / N, [cx, 0]), q.moveTo([x0 + (L * k) / N, r * Math.sin(th)]), { duration: 0.1, ease: "linear" });
}
q.moveTo([x0, 0]);
