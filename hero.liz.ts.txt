// 販売サイトの最初の面の図（site/hero.mp4）。Liz CAP でこのまま書き出したもの。
// ローレンツ方程式の軌跡を、点の集まりと細い線で描き、カメラを 1 周させる（12 秒で始まりにつながる）。

const SIGMA = 10, RHO = 28, BETA = 8 / 3;
const INK = "#d4d4d8", SIGNAL = "#9cc3ff";
background("#0b0b0c");

// ルンゲ＝クッタ法で軌跡を計算し、画面の大きさに縮める
const pts: Vec3[] = [];
let [x, y, z] = [0.1, 0, 20];
const f = (x: number, y: number, z: number): Vec3 => [SIGMA * (y - x), x * (RHO - z) - y, x * y - BETA * z];
const dt = 0.005, S = 0.16;
for (let i = 0; i < 9000; i++) {
  const k1 = f(x, y, z);
  const k2 = f(x + (dt / 2) * k1[0], y + (dt / 2) * k1[1], z + (dt / 2) * k1[2]);
  const k3 = f(x + (dt / 2) * k2[0], y + (dt / 2) * k2[1], z + (dt / 2) * k2[2]);
  const k4 = f(x + dt * k3[0], y + dt * k3[1], z + dt * k3[2]);
  x += (dt / 6) * (k1[0] + 2 * k2[0] + 2 * k3[0] + k4[0]);
  y += (dt / 6) * (k1[1] + 2 * k2[1] + 2 * k3[1] + k4[1]);
  z += (dt / 6) * (k1[2] + 2 * k2[2] + 2 * k3[2] + k4[2]);
  if (i > 300) pts.push([x * S, y * S, (z - 25) * S]);
}

setCamera({ phi: 72 * DEGREES, theta: -90 * DEGREES, zoom: 0.92 });
const n = pts.length - 1;
curve3d((t) => pts[Math.round(t)], { t: [0, n], samples: 4000, color: INK, width: 0.008, opacity: 0.45 });
for (let i = 0; i < n; i += 7) {
  const [px, py, pz] = pts[i];
  const mark = i % 350 === 0;
  dot3d({ x: px, y: py, z: pz, r: mark ? 0.03 : 0.013, color: mark ? SIGNAL : INK, opacity: mark ? 0.95 : 0.7 });
}
// 足もとの目盛りの輪と、z 軸（計測の図のように）
const floor = -3.4;
for (const R of [2.2, 3.4, 4.6]) curve3d((t) => [R * Math.cos(t), R * Math.sin(t), floor], { t: [0, TAU], samples: 120, color: INK, width: 0.006, opacity: 0.18 });
line3d({ from: [0, 0, floor], to: [0, 0, 3.6], color: SIGNAL, width: 0.006, opacity: 0.35 });
line3d({ from: [-4.6, 0, floor], to: [4.6, 0, floor], color: INK, width: 0.006, opacity: 0.15 });
line3d({ from: [0, -4.6, floor], to: [0, 4.6, floor], color: INK, width: 0.006, opacity: 0.15 });
// 2 つの平衡点（渦の中心）に青い印
const c = Math.sqrt(BETA * (RHO - 1));
dot3d({ x: c * S, y: c * S, z: (RHO - 1 - 25) * S, r: 0.05, color: SIGNAL });
dot3d({ x: -c * S, y: -c * S, z: (RHO - 1 - 25) * S, r: 0.05, color: SIGNAL });

play(orbit(TAU), { duration: 12, ease: "linear" });
