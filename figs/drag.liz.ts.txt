// 販売サイト「できること 02 つかんで動かすと、コードが変わる」の図（site/figs/drag.mp4）。
// 点をつかんで動かすと、座標と下のコードの数字がいっしょに変わる。最後は始まりの位置に戻る。

const INK = "#d4d4d8", DIM = "#2a2a2e", SIGNAL = "#9cc3ff", FOG = "#9a9aa0";
background("#0b0b0c");

const plane = numberPlane({ xRange: [-7, 7, 1], yRange: [-3, 3, 1], width: 12, height: 5.2, y: 0.3, color: DIM, axisColor: "#3a3a3f" });
const p = dot({ x: plane.c2p(1, 1)[0], y: plane.c2p(1, 1)[1], r: 0.11, color: SIGNAL });
const toUnit = () => { const [x, y] = p.position(); const o = plane.c2p(0, 0), e = plane.c2p(1, 1); return [(x - o[0]) / (e[0] - o[0]), (y - o[1]) / (e[1] - o[1])]; };

always(() => {
  const [x, y] = p.position();
  const o = plane.c2p(0, 0);
  const [ux, uy] = toUnit();
  const guides: Handle[] = [];   // 軸までの破線（短すぎるときは描かない）
  if (Math.abs(y - o[1]) > 0.3) guides.push(dashedLine({ from: [x, y], to: [x, o[1]], color: SIGNAL, width: 0.015, opacity: 0.6 }));
  if (Math.abs(x - o[0]) > 0.3) guides.push(dashedLine({ from: [x, y], to: [o[0], y], color: SIGNAL, width: 0.015, opacity: 0.6 }));
  return group(
    ...guides,
    // つまんでいる手（矢印のカーソル）
    polygon({ points: [[x + 0.12, y - 0.1], [x + 0.12, y - 0.75], [x + 0.28, y - 0.6], [x + 0.42, y - 0.88], [x + 0.52, y - 0.83], [x + 0.39, y - 0.55], [x + 0.6, y - 0.55]], fill: INK, stroke: "#0b0b0c", strokeWidth: 0.02 }),
    text(`(${ux.toFixed(1)}, ${uy.toFixed(1)})`, { x: x + 1.15, y: y + 0.35, size: 0.3, color: INK, font: "JetBrains Mono" }),
  );
});

// 下のコード
line({ from: [-6, -2.75], to: [6, -2.75], color: "#3a3a3f", width: 0.01 });
always(() => {
  const [ux, uy] = toUnit();
  return text(`dot({ x: ${ux.toFixed(1)}, y: ${uy.toFixed(1)} })`, { x: -3.2, y: -3.25, size: 0.34, color: INK, font: "JetBrains Mono" });
});
text("DRAG → CODE", { x: 5.0, y: -3.25, size: 0.26, color: FOG, font: "JetBrains Mono" });

wait(0.5);
play(p.moveTo(plane.c2p(4, 2)), { duration: 1.4 });
play(p.moveTo(plane.c2p(3, -2)), { duration: 1.2 });
play(p.moveTo(plane.c2p(-4, -1)), { duration: 1.6 });
play(p.moveTo(plane.c2p(-2, 2)), { duration: 1.2 });
play(p.moveTo(plane.c2p(1, 1)), { duration: 1.3 });
wait(0.5);
